        // 2. Tab Navigation Logic
        function switchTab(tabId) {
            document.querySelectorAll('.module-content').forEach(el => el.classList.remove('active'));

        // 3. Sandbox Drag & Drop Logic (Module A)
        let draggedComponentId = null;

        function drag(ev, symId) {
            ev.dataTransfer.setData("text", symId);
        }

        function allowDrop(ev) {
            ev.preventDefault();
        }

        function drop(ev) {
            ev.preventDefault();
            const symId = ev.dataTransfer.getData("text");
            
            // Create a new SVG element dynamically on drop
            const newSvg = document.createElementNS("http://www.w3.org/2000/svg", "svg");
            newSvg.setAttribute("class", "draggable-svg");
            newSvg.style.position = "absolute";
            
            // Adjust position based on drop coordinates
            const rect = ev.target.getBoundingClientRect();
            let x = ev.clientX - rect.left - 25; // center the 50x50 icon
            let y = ev.clientY - rect.top - 25;
            
            // Snap to grid (20px grid size defined in CSS)
            x = Math.round(x / 20) * 20;
            y = Math.round(y / 20) * 20;

            newSvg.style.left = x + "px";
            newSvg.style.top = y + "px";

            const useTag = document.createElementNS("http://www.w3.org/2000/svg", "use");
            useTag.setAttribute("href", "#" + symId);
            newSvg.appendChild(useTag);

            // Allow dragging inside the canvas
            newSvg.onmousedown = function(e) {
                let shiftX = e.clientX - newSvg.getBoundingClientRect().left;
                let shiftY = e.clientY - newSvg.getBoundingClientRect().top;

                function moveAt(pageX, pageY) {
                    let newX = pageX - rect.left - shiftX;
                    let newY = pageY - rect.top - shiftY;
                    newSvg.style.left = Math.round(newX / 20) * 20 + 'px';
                    newSvg.style.top = Math.round(newY / 20) * 20 + 'px';
                }

                function onMouseMove(event) { moveAt(event.pageX, event.pageY); }
                document.addEventListener('mousemove', onMouseMove);
                
                newSvg.onmouseup = function() {
                    document.removeEventListener('mousemove', onMouseMove);
                    newSvg.onmouseup = null;
                };
            };
            newSvg.ondragstart = function() { return false; };

            document.getElementById("dropZone").appendChild(newSvg);
        }

        // 4. Lab Investigation Logic (Module B)
        const voltageConstant = 5.0; // V
        
        function updateLab() {
            const d = parseFloat(document.getElementById('distSlider').value);
            document.getElementById('distVal').innerText = d.toFixed(1);
            
            // Simulated Physics: R = base_resistance + (resistivity * distance)
            const R = 2.0 + (1.5 * d); 
            const I = voltageConstant / R;
            
            document.getElementById('currentVal').innerText = I.toFixed(2);
        }

        function recordData() {
            const d = parseFloat(document.getElementById('distSlider').value).toFixed(1);
            const I = parseFloat(document.getElementById('currentVal').innerText).toFixed(2);
            const V = voltageConstant.toFixed(1);
            const R = (voltageConstant / I).toFixed(1);

            const tbody = document.querySelector("#labTable tbody");
            const row = `<tr>
                <td>${d}</td>
                <td>${I}</td>
                <td>${V}</td>
                <td>${R}</td>
            </tr>`;
            tbody.insertAdjacentHTML('beforeend', row);
        }

        // 5. Transformer Calculator Logic (Module C)
        function calcTransformer() {
            const Vp = document.getElementById('Vp');
            const Np = document.getElementById('Np');
            const Vs = document.getElementById('Vs');
            const Ns = document.getElementById('Ns');
            
            const v_Vp = parseFloat(Vp.value);
            const v_Np = parseFloat(Np.value);
            const v_Vs = parseFloat(Vs.value);
            const v_Ns = parseFloat(Ns.value);

            let resultText = "";

            if (!Vp.value && Np.value && Vs.value && Ns.value) {
                Vp.value = ((v_Vs * v_Np) / v_Ns).toFixed(2);
                resultText = `Calculated Vp = ${Vp.value} V`;
            } else if (!Vs.value && Np.value && Vp.value && Ns.value) {
                Vs.value = ((v_Vp * v_Ns) / v_Np).toFixed(2);
                resultText = `Calculated Vs = ${Vs.value} V`;
            } else if (!Np.value && Vp.value && Vs.value && Ns.value) {
                Np.value = Math.round((v_Vp * v_Ns) / v_Vs);
                resultText = `Calculated Np = ${Np.value} Turns`;
            } else if (!Ns.value && Vp.value && Vs.value && Np.value) {
                Ns.value = Math.round((v_Vs * v_Np) / v_Vp);
                resultText = `Calculated Ns = ${Ns.value} Turns`;
            } else {
                resultText = "Harap isi tepat 3 kotak dan biarkan 1 kotak kosong untuk dihitung.";
            }

            document.getElementById('calcResult').innerText = resultText;
            document.getElementById('calcResult').style.color = "var(--primary-color)";
        }

        // 6. Practice Hub Logic (Module D)
        function checkQ1() {
            const selected = document.querySelector('input[name="q1"]:checked');
